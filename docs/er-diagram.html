My Bad, Coach — SQLite data model

CRUD tables (users → teams → events → matches, players, roster) plus the append-only match event log. Stats are derived, not stored.

erDiagram
    users ||--o{ team_memberships : "belongs to"
    teams ||--o{ team_memberships : "has"
    teams ||--o{ events : "organizes"
    events ||--o{ matches : "contains"
    teams ||--o{ players : "has"
    players ||--o{ roster_entries : "appears in"
    matches ||--o{ roster_entries : "fields"
    matches ||--o{ match_events : "logs"

    users {
        uuid id PK
        text email UK
        text name
        datetime created_at
    }

    teams {
        uuid id PK
        text name
        datetime created_at
    }

    team_memberships {
        uuid id PK
        uuid user_id FK
        uuid team_id FK
        text role
        datetime created_at
    }

    events {
        uuid id PK
        uuid team_id FK
        text name
        date start_date
        date end_date
        text location
    }

    matches {
        uuid id PK
        uuid event_id FK
        text opponent_name
        datetime scheduled_at
        int point_cap
        datetime created_at
    }

    players {
        uuid id PK
        uuid team_id FK
        text name
        int jersey_number
        datetime created_at
    }

    roster_entries {
        uuid id PK
        uuid match_id FK
        uuid player_id FK
        bool is_starting
        int position
        datetime created_at
    }

    match_events {
        uuid id PK
        uuid match_id FK
        int point_number
        bigint effective_ts
        text type
        json payload
        datetime created_at
    }
